def kpi_card(title: str, value, sub: str = ""):
    sub_html = f'<div class="kpi-sub">{sub}</div>' if sub else ""
    return f"""
    <div class="kpi-card">
        <div class="kpi-title">{title}</div>
        <div class="kpi-value">{value}</div>
        {sub_html}
    </div>
    """


def panel_header(title: str, desc: str = ""):
    desc_html = f'<span class="panel-desc">{desc}</span>' if desc else ""
    st.markdown(
        f"""
        <div class="panel-header">
            <p class="panel-title">{title}</p>
            {desc_html}
        </div>
        """,
        unsafe_allow_html=True,
    )


def apply_chart_theme(fig, height=380):
    """Apply one consistent look to every Plotly figure in the app."""
    fig.update_layout(
        template="plotly_white",
        font=dict(family=FONT, size=12.5, color=COLORS["text"]),
        colorway=CHART_PALETTE,
        height=height,
        margin=dict(l=10, r=10, t=30, b=10),
        plot_bgcolor="rgba(0,0,0,0)",
        paper_bgcolor="rgba(0,0,0,0)",
        legend=dict(orientation="h", yanchor="bottom", y=1.02, xanchor="right", x=1),
    )
    fig.update_xaxes(gridcolor="#eef0f2", zeroline=False)
    fig.update_yaxes(gridcolor="#eef0f2", zeroline=False)
    return fig
